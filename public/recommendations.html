<script type="module">

    // 初始化 Firebase

    let app;

    let db;


    try {

        if (firebaseConfig.apiKey) {

            app = initializeApp(firebaseConfig);

            db = getFirestore(app);

        } else {

            console.warn("尚未設定 firebaseConfig。");

        }

    } catch (e) {

        console.error("Firebase 初始化失敗", e);

    }


    // 假定用戶的品味 Profile 標籤 (硬編碼，稍後會修改成讀取)

    // 這是我們用來匹配 POI 的核心標籤

    const USER_PREFS_TAGS = ['ACG-科幻', '現代藝術']; 


    const searchBtn = document.getElementById('searchBtn');

    const locationInput = document.getElementById('locationInput');

    const recommendationsList = document.getElementById('recommendationsList');

    const statusMessage = document.getElementById('statusMessage');


    // --- 核心推薦邏輯 ---

    async function fetchAndRecommend(location) {

        recommendationsList.innerHTML = '';

        statusMessage.innerHTML = '搜尋中...';

        searchBtn.disabled = true;


        if (!db) {

            statusMessage.innerHTML = '<span class="error">錯誤：資料庫未連接。</span>';

            searchBtn.disabled = false;

            return;

        }


        try {

            // 1. 建立查詢：先篩選地點

            // 我們假設 POI 集合的名稱是 'Points_of_Interest' (需要您在 Firebase 中建立)

            const poiCollection = collection(db, 'Points_of_Interest');

            

            // 創建查詢：只搜尋地點相符的 POI

            const locationQuery = query(

                poiCollection,

                where('地點', '==', location) // 欄位名稱要與您的 POI 集合完全匹配

            );


            const querySnapshot = await getDocs(locationQuery);

            const allLocationPOIs = [];


            querySnapshot.forEach((doc) => {

                allLocationPOIs.push({ id: doc.id, ...doc.data() });

            });


            // 2. 實作標籤匹配邏輯

            const recommendedPOIs = [];

            

            allLocationPOIs.forEach(poi => {

                // 檢查 POI 的 '內容標籤' 陣列中，是否包含任一 'USER_PREFS_TAGS' 中的標籤

                let matchCount = 0;


                // 確保 POI 內容標籤是一個陣列

                if (Array.isArray(poi.內容標籤)) {

                    poi.內容標籤.forEach(poiTag => {

                        if (USER_PREFS_TAGS.includes(poiTag)) {

                            matchCount++;

                        }

                    });

                }

                

                // 如果至少匹配到一個標籤，則將其加入推薦列表

                if (matchCount > 0) {

                    recommendedPOIs.push({ 

                        ...poi, 

                        matchScore: matchCount // 簡單地以匹配標籤數量作為分數

                    });

                }

            });


            // 3. 排序與顯示結果

            // 依匹配分數由高到低排序

            recommendedPOIs.sort((a, b) => b.matchScore - a.matchScore);


            if (recommendedPOIs.length === 0) {

                statusMessage.innerHTML = `<span style="color:#ff6600;">在 ${location} 找不到符合您品味 (${USER_PREFS_TAGS.join(', ')}) 的活動/景點。</span>`;

            } else {

                recommendedPOIs.forEach(poi => {

                    const listItem = document.createElement('li');

                    listItem.className = 'recommendation-item';

                    listItem.innerHTML = `

                        <h3>${poi.名稱} (${poi.地點})</h3>

                        <p class="tags">標籤: ${poi.內容標籤 ? poi.內容標籤.join(', ') : '無'}</p>

                        <p class="match-score">品味匹配度: ${poi.matchScore} 個標籤</p>

                    `;

                    recommendationsList.appendChild(listItem);

                });


                statusMessage.innerHTML = `<span class="success">成功找到 ${recommendedPOIs.length} 個推薦項目！</span>`;

            }


        } catch (error) {

            console.error("搜尋推薦時發生錯誤: ", error);

            statusMessage.innerHTML = `<span class="error">資料讀取失敗，請檢查 Firestore 連線和規則。</span>`;

        } finally {

            searchBtn.disabled = false;

            searchBtn.textContent = '重新推薦';

        }

    }


    // 事件監聽：按下按鈕時觸發推薦

    searchBtn.addEventListener('click', () => {

        const location = locationInput.value.trim();

        if (location) {

            fetchAndRecommend(location);

        } else {

            statusMessage.innerHTML = '<span class="error">請輸入旅遊地點！</span>';

        }

    });


    // 頁面加載後自動執行一次

    fetchAndRecommend(locationInput.value.trim());


</script>
